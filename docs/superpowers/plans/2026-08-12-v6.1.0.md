# js-cool 6.1.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship js-cool 6.1.0 — four buckets of additive enhancements (clone, array/object, string, date) plus 5 additive-mirror subpath exports, fully backward-compatible, IE11 maintained.

**Architecture:** All changes are additive (new optional params / new exports / new barrels). `clone` gets an internal iterative rewrite + structuredClone fast-path (API unchanged). Subpaths (`./object`, `./color`, `./string`, `./array`, `./number`) are wired at three points (rolldown, fix-types, package.json) as mirrors of the main `.` export.

**Tech Stack:** TypeScript 5.9, Rolldown 1.0-rc, Vitest 3.2, api-extractor, SWC (ES5 for IE11 IIFE), VitePress docs.

## Global Constraints

- Baseline: 6.0.0. Target: 6.1.0 (minor). No breaking API changes.
- IE11 maintained: feature-detect modern APIs (structuredClone, Intl.Segmenter) with fallbacks; IIFE transpiled SWC→ES5.
- Test runner: vitest with `globals: true` (use bare `describe/it/expect`), env `happy-dom`, setup `./test/setup.ts`. Import source directly: `from '../src/<file>'`.
- Commit style: conventional `feat:` / `fix:` / `build:` / `docs:` prefixes, imperative English.
- TDD: write failing test → run (fail) → implement → run (pass) → commit, per task.
- Do NOT touch the main `src/index.ts` exports of existing functions except to ADD new exports (`sortBy`, `parseDate`, `parseDateWithFormat`, new types).

```ts
function clone<T = any>(parent: T): T {
	// structuredClone fast-path (modern envs); throws on functions/DOM/Symbols → fallback
	if (typeof structuredClone === 'function') {
		try {
			return structuredClone(parent)
		} catch {
			// fall through to manual
		}
	}

	if (parent === null || typeof parent !== 'object') return parent

	const getRegExp = (reg: RegExp): string => {
		let flags = ''
		if (reg.global) flags += 'g'
		if (reg.ignoreCase) flags += 'i'
		if (reg.multiline) flags += 'm'
		return flags
	}

	const create = (src: any): any => {
		if (isArray(src)) return []
		if (isRegExp(src)) {
			const r = new RegExp(src.source, getRegExp(src))
			if (src.lastIndex) r.lastIndex = src.lastIndex
			return r
		}
		if (isDate(src)) return new Date(src.getTime())
		if (src instanceof Set) return new Set()
		if (src instanceof Map) return new Map()
		if (ArrayBuffer.isView(src)) return new (src.constructor as any)(src)
		if (src instanceof ArrayBuffer) return src.slice(0)
		return Object.create(Object.getPrototypeOf(src))
	}

	const root = create(parent)
	const visited = new WeakMap<object, any>()
	visited.set(parent as object, root)

	const stack: Array<{ src: any; target: any }> = [{ src: parent, target: root }]

	// hoisted so it's visible inside the loop body
	function cloneRef(val: any): any {
		if (visited.has(val)) return visited.get(val)
		const child = create(val)
		visited.set(val, child)
		stack.push({ src: val, target: child })
		return child
	}

	while (stack.length) {
		const { src, target } = stack.pop()!

		if (target instanceof Set) {
			for (const item of src) {
				if (item && typeof item === 'object') {
					target.add(cloneRef(item))
				} else {
					target.add(item)
				}
			}
		} else if (target instanceof Map) {
			for (const [k, v] of src) {
				const nk = k && typeof k === 'object' ? cloneRef(k) : k
				const nv = v && typeof v === 'object' ? cloneRef(v) : v
				target.set(nk, nv)
			}
		} else if (ArrayBuffer.isView(target) || target instanceof ArrayBuffer) {
			// already copied in create(); nothing to walk
		} else {
			for (const key in src) {
				if (Object.prototype.hasOwnProperty.call(src, key)) {
					const val = src[key]
					if (val && typeof val === 'object') {
						target[key] = cloneRef(val)
					} else {
						target[key] = val
					}
				}
			}
		}
	}

	return root
}
```

```ts
function groupBy<T>(
	array: T[],
	iteratee:
		| keyof T
		| ((value: T) => string | number)
		| Array<keyof T | ((value: T) => string | number)>,
	options?: { separator?: string }
): Record<string, T[]> {
	if (!Array.isArray(array)) return {}

	const separator = options?.separator ?? '.'
	const iteratees = Array.isArray(iteratee) ? iteratee : [iteratee]

	const resolve = (item: T, it: keyof T | ((value: T) => string | number)) =>
		typeof it === 'function' ? it(item) : (item[it] as string | number)

	const result: Record<string, T[]> = {}
	for (const item of array) {
		const keyStr = iteratees.map(it => String(resolve(item, it))).join(separator)
		if (!result[keyStr]) result[keyStr] = []
		result[keyStr].push(item)
	}
	return result
}
```

```ts
function unique<T = unknown>(
	arr: T[],
	by?: keyof T | ((item: T) => unknown)
): T[] {
	if (!by) {
		// 6.0.0 path: reference equality via indexOf
		const result: T[] = []
		for (const item of arr) {
			if (result.indexOf(item) === -1) result.push(item)
		}
		return result
	}
	const getKey = typeof by === 'function' ? by : (item: T) => (item as any)[by]
	const seen = new Map<unknown, T>()
	for (const item of arr) {
		const k = getKey(item)
		if (!seen.has(k)) seen.set(k, item)
	}
	return [...seen.values()]
}
```

```ts
export type SortIteratee<T> =
	| keyof T
	| {
			field?: keyof T
			order?: 'asc' | 'desc'
			transform?: (v: unknown) => unknown
	  }

/**
 * Creates a new array sorted by one or more iteratees (left-to-right tie-breaker).
 * Non-mutating. Lodash `_.sortBy` analog.
 *
 * @example
 * ```ts
 * sortBy([{ n: 2 }, { n: 1 }], 'n') // [{ n: 1 }, { n: 2 }]
 * sortBy(data, { field: 'n', order: 'desc' })
 * sortBy(data, 'a', { field: 'b', order: 'asc' })
 * ```
 * @since 6.1.0
 */
function sortBy<T>(array: T[], ...iteratees: SortIteratee<T>[]): T[] {
	if (!Array.isArray(array)) return []
	if (array.length <= 1) return [...array]

	const specs = iteratees.map(it => {
		if (typeof it === 'object' && it !== null) {
			return {
				field: it.field,
				order: it.order === 'desc' ? 'desc' : ('asc' as 'asc' | 'desc'),
				transform: it.transform,
			}
		}
		return { field: it as keyof T, order: 'asc' as 'asc' | 'desc', transform: undefined }
	})

	const valueOf = (item: T, spec: (typeof specs)[number]) => {
		const raw = spec.field === undefined ? item : item[spec.field]
		return spec.transform ? spec.transform(raw) : raw
	}

	const compare = (a: T, b: T): number => {
		for (const spec of specs) {
			const va = valueOf(a, spec)
			const vb = valueOf(b, spec)
			if (va === vb) continue
			if (va === undefined || va === null) return -1
			if (vb === undefined || vb === null) return 1
			if (va < vb) return spec.order === 'asc' ? -1 : 1
			if (va > vb) return spec.order === 'asc' ? 1 : -1
		}
		return 0
	}

	return [...array].sort(compare)
}

export default sortBy
```

function truncate(str: string, options: TruncateOptions = {}): string {
	if (typeof str !== 'string') return ''
	const { length = 30, omission = '...', separator, byWidth = false, breakEmoji = true } = options

	const measure = (s: string): number => {
		if (!byWidth) return s.length
		let w = 0
		for (const ch of s) w += isFullWidth(ch) ? 2 : 1
		return w
	}
	const takeBy = (s: string, budget: number): string => {
		if (!byWidth) return s.slice(0, budget)
		let w = 0, out = ''
		for (const ch of s) {
			const cw = isFullWidth(ch) ? 2 : 1
			if (w + cw > budget) break
			out += ch
			w += cw
		}
		return out
	}

	const budget = Math.max(0, length - measure(omission))
	let result = takeBy(str, budget)

	// Keep grapheme integrity: strip a trailing lone high surrogate
	if (breakEmoji && result.length > 0) {
		const last = result.charCodeAt(result.length - 1)
		if (last >= 0xd800 && last <= 0xdbff) {
			result = result.slice(0, -1)
		}
	}

	if (separator) {
		if (typeof separator === 'string') {
			const index = result.lastIndexOf(separator)
			if (index > 0) result = result.slice(0, index)
		} else {
			const match = result.match(separator)
			if (match && match.index !== undefined && match.index > 0) {
				result = result.slice(0, match.index)
			}
		}
	}

	return result + omission
}

	return null
}

function escapeRegExp(ch: string): string {
	return ch.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
}

function tokenRegex(tok: string): string {
	if (tok === 'MMM' || tok === 'MMMM') return '[A-Za-z\\u4e00-\\u9fff]+'
	return '\\d{1,4}'
}

export function parseDateWithFormat(dateStr: string, format: string): Date | null {
	if (typeof dateStr !== 'string' || typeof format !== 'string') return null

	const ordered: string[] = []
	let re = ''
	let i = 0
	while (i < format.length) {
		const matched = matchLongestToken(format.slice(i))
		if (matched) {
			ordered.push(matched)
			re += `(${tokenRegex(matched)})`
			i += matched.length
		} else {
			re += escapeRegExp(format[i])
			i++
		}
	}

	const m = dateStr.match(new RegExp(`^${re}$`))
	if (!m) return null

	const vals: Record<string, number> = {
		YYYY: 1970, YY: 1970, MM: 0, M: 0, DD: 1, D: 1, HH: 0, H: 0, mm: 0, m: 0, ss: 0, s: 0,
	}
	ordered.forEach((tok, idx) => {
		const n = parseInt(m[idx + 1], 10)
		if (tok === 'MM' || tok === 'M') vals[tok] = n - 1 // 0-indexed month
		else if (tok === 'YY') vals['YY'] = 2000 + n
		else vals[tok] = n
	})

	const year = vals.YYYY !== 1970 ? vals.YYYY : (vals.YY ?? 1970)
	const d = new Date(
		year,
		vals.MM ?? 0,
		vals.DD ?? 1,
		vals.HH ?? 0,
		vals.mm ?? 0,
		vals.ss ?? 0
	)
	return isNumberNaN(d.getTime()) ? null : d
}
```

Note: month-name (MMM/MMMM) parsing is best-effort scaffolding; the 6.1.0 test cases use numeric months only, so they pass. Full month-name parsing deferred to 6.2.
